cmake_minimum_required(VERSION 3.20)
project(nexusretro_worker LANGUAGES CXX)
add_executable(retro-worker src/main.cpp)
target_compile_features(retro-worker PRIVATE cxx_std_20)
target_include_directories(retro-worker PRIVATE vendor)
target_link_libraries(retro-worker PRIVATE ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(retro-worker PRIVATE /W4)
else()
  target_compile_options(retro-worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
